@[reducible] def Domino.Cell : ℕ → ℕ → Type :=
fun m w => Fin m × Fin w
def Domino.Adjacent : {m w : ℕ} → Domino.Cell m w → Domino.Cell m w → Prop :=
fun {m w} a b => a.1 = b.1 ∧ (↑a.2 + 1 = ↑b.2 ∨ ↑b.2 + 1 = ↑a.2) ∨ a.2 = b.2 ∧ (↑a.1 + 1 = ↑b.1 ∨ ↑b.1 + 1 = ↑a.1)
def Domino.IsEdge : {m w : ℕ} → Finset (Domino.Cell m w) → Prop :=
fun {m w} s => ∃ a b, Domino.Adjacent a b ∧ s = {a, b}
def Domino.IsCover : {m w : ℕ} → Finset (Domino.Cell m w) → Finset (Domino.Edge m w) → Prop :=
fun {m w} holes edges =>
  (∀ (c : Domino.Cell m w), (∃ e ∈ edges, c ∈ ↑e) ↔ c ∉ holes) ∧ ∀ e ∈ edges, ∀ f ∈ edges, ∀ c ∈ ↑e, c ∈ ↑f → e = f
@[reducible] def Domino.Tiling : ℕ → ℕ → Type :=
fun m w => Domino.Cover m w ∅
@[reducible] def Domino.PartialTiling : (m : ℕ) → ℕ → Domino.State m → Type :=
fun m w S => { _T // w = 0 → S = ∅ }
structure Domino.RectangleSymmetry : Type
number of parameters: 0
fields:
  Domino.RectangleSymmetry.flipRows : Bool
  Domino.RectangleSymmetry.flipCols : Bool
constructor:
  Domino.RectangleSymmetry.mk (flipRows flipCols : Bool) : Domino.RectangleSymmetry
def Domino.cellAction : {m w : ℕ} → Domino.RectangleSymmetry → Domino.Cell m w → Domino.Cell m w :=
fun {m w} g c => (if g.flipRows = true then c.1.rev else c.1, if g.flipCols = true then c.2.rev else c.2)
@[reducible] def Domino.TilingOrbit : ℕ → ℕ → Type :=
fun m w => Quotient (MulAction.orbitRel Domino.RectangleSymmetry (Domino.Tiling m w))
def Domino.ordinaryCount : ℕ → ℕ → ℕ :=
fun m n => Fintype.card (Domino.Tiling m (2 * n))
def Domino.fixedCount : ℕ → ℕ → Domino.RectangleSymmetry → ℕ :=
fun m n g => Fintype.card { T // g • T = T }
def Domino.orbitCount : ℕ → ℕ → ℕ :=
fun m n => Fintype.card (Domino.TilingOrbit m (2 * n))
def Domino.a : ℕ → ℕ :=
fun n => Domino.orbitCount 3 n
def Domino.Legal3 : Domino.State 3 → Domino.State 3 → Prop :=
fun p q => Disjoint p q ∧ (Finset.univ \ (p ∪ q) = ∅ ∨ Finset.univ \ (p ∪ q) = {0, 1} ∨ Finset.univ \ (p ∪ q) = {1, 2})
